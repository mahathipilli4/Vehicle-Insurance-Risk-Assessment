from flask import (
    Flask, render_template, request,
    redirect, session, url_for
)
import sqlite3
import joblib
import pandas as pd
from pathlib import Path
from functools import wraps
from werkzeug.security import check_password_hash
import os

app = Flask(__name__)

app.secret_key = os.environ.get(
    "FLASK_SECRET_KEY",
    "development-only-change-this-secret"
)

BASE_DIR = Path(__file__).resolve().parent

# Load trained model
model = joblib.load(BASE_DIR / "vehicle_insurance_model.pkl")

FEATURES = [
    "driver_age",
    "past_accidents",
    "speeding_tickets",
    "annual_mileage"
]


# Connect to existing users database
def get_db_connection():
    conn = sqlite3.connect(BASE_DIR / "users.db")
    conn.row_factory = sqlite3.Row
    return conn


# Check login status
def login_required(view_function):
    @wraps(view_function)
    def wrapped(*args, **kwargs):
        if "username" not in session or "role" not in session:
            return redirect(url_for("login"))
        return view_function(*args, **kwargs)
    return wrapped


# Login page
@app.route("/", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        role = request.form.get("role", "").strip()
        username = request.form.get("username", "").strip()
        password = request.form.get("password", "")

        conn = get_db_connection()

        try:
            account = conn.execute(
                """
                SELECT username, password, role
                FROM users
                WHERE username = ? AND role = ?
                """,
                (username, role)
            ).fetchone()
        finally:
            conn.close()

        if account and check_password_hash(
            account["password"], password
        ):
            session.clear()
            session["username"] = account["username"]
            session["role"] = account["role"]

            return redirect(url_for("dashboard"))

        return render_template(
            "login.html",
            error="Invalid role, username, or password."
        ), 401

    if "username" in session:
        return redirect(url_for("dashboard"))

    return render_template("login.html")


# Dashboard

# Predict claim probability
@app.route("/predict", methods=["POST"])
@login_required
def predict():
    try:
        driver_age = int(request.form["driver_age"])
        past_accidents = int(request.form["past_accidents"])
        speeding_tickets = int(request.form["speeding_tickets"])
        annual_mileage = float(request.form["annual_mileage"])

        # Validate inputs
        if not 18 <= driver_age <= 100:
            return "Driver age must be between 18 and 100.", 400

        if (
            past_accidents < 0
            or speeding_tickets < 0
            or annual_mileage < 0
        ):
            return "Values cannot be negative.", 400

        # Prepare model input
        input_data = pd.DataFrame(
            [[
                driver_age,
                past_accidents,
                speeding_tickets,
                annual_mileage
            ]],
            columns=FEATURES
        )

        # Predict probability
        probabilities = model.predict_proba(input_data)[0]
        classes = list(model.classes_)

        if 1 not in classes:
            raise ValueError("Claim class 1 is missing.")

        claim_probability = (
            float(probabilities[classes.index(1)]) * 100
        )

        probability = round(claim_probability, 2)
        prediction_value = int(model.predict(input_data)[0])

        # Categorize estimated risk
        if claim_probability <= 30:
            risk_label = "Low Claim Risk"
        elif claim_probability <= 60:
            risk_label = "Medium Claim Risk"
        else:
            risk_label = "High Claim Risk"

        risk_description = (
            "This is a model-estimated claim probability, "
            "not a guarantee of a future claim or an actual "
            "insurance decision."
        )

        # Save assessment history
        history_file = BASE_DIR / "assessment_history.csv"

        record = pd.DataFrame([{
            "driver_age": driver_age,
            "past_accidents": past_accidents,
            "speeding_tickets": speeding_tickets,
            "annual_mileage": annual_mileage,
            "prediction": prediction_value,
            "claim_probability": probability
        }])

        record.to_csv(
            history_file,
            mode="a",
            header=not history_file.exists()
                or history_file.stat().st_size == 0,
            index=False
        )

        return render_template(
            "result.html",
            driver_age=driver_age,
            past_accidents=past_accidents,
            speeding_tickets=speeding_tickets,
            annual_mileage=annual_mileage,
            risk_label=risk_label,
            risk_description=risk_description,
            probability=probability,
            role=session["role"]
        )

    except (ValueError, KeyError):
        return "Please enter valid values in all fields.", 400

@app.route("/history")
@login_required
def history():
    history_file = BASE_DIR / "assessment_history.csv"

    if history_file.exists() and history_file.stat().st_size > 0:
        df = pd.read_csv(history_file).fillna("")

        "history.html",

        username=session["username"],
        records=records

import pandas as pd
import joblib
from pathlib import Path

from sklearn.pipeline import Pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score,
    precision_score,
    recall_score,
    f1_score,
    roc_auc_score,
    confusion_matrix,
    classification_report
)

folder = Path(__file__).resolve().parent

# Load train and test datasets
train_df = pd.read_csv(folder / "X_train.csv")
test_df = pd.read_csv(folder / "X_test.csv")

X_train = train_df.drop(columns=["claim_filed"])
y_train = train_df["claim_filed"]

X_test = test_df.drop(columns=["claim_filed"])
y_test = test_df["claim_filed"]

# Logistic Regression with feature scaling
model = Pipeline([
    ("scaler", StandardScaler()),
    ("classifier", LogisticRegression(
        max_iter=1000,
        random_state=42
    ))
])

# Train the model
model.fit(X_train, y_train)

# Predictions
y_pred = model.predict(X_test)
y_prob = model.predict_proba(X_test)[:, 1]

# Performance metrics
print("\n===== LOGISTIC REGRESSION PERFORMANCE =====")
print(f"Accuracy  : {accuracy_score(y_test, y_pred):.4f}")
print(f"Precision : {precision_score(y_test, y_pred, zero_division=0):.4f}")
print(f"Recall    : {recall_score(y_test, y_pred, zero_division=0):.4f}")
print(f"F1-Score  : {f1_score(y_test, y_pred, zero_division=0):.4f}")
print(f"ROC-AUC   : {roc_auc_score(y_test, y_prob):.4f}")

print("\n===== CONFUSION MATRIX =====")
print(confusion_matrix(y_test, y_pred))

print("\n===== CLASSIFICATION REPORT =====")
print(classification_report(
    y_test,
    y_pred,
    target_names=["No Claim (0)", "Claim Filed (1)"],
    zero_division=0
))

# Save the trained model
joblib.dump(model, folder / "vehicle_insurance_model.pkl")

print("\nLogistic Regression model saved successfully!")
